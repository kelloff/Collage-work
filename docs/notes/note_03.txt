PYTHON — Записка 3. Функции, списки, вложенные циклы
(под задания уровней 2–3: def, return, sum, таблица умножения)

══════════════════════════════════════
ФУНКЦИИ def И return
══════════════════════════════════════
def mul(a, b):
	return a * b

print(mul(3, 4))

# Сумма элементов списка:
def total(nums):
	s = 0
	for x in nums:
		s += x
	return s

print(total([1, 2, 3, 4]))

# Короче встроенной функцией:
print(sum([1, 2, 3, 4]))

══════════════════════════════════════
СУММА ЧИСЕЛ ОТ 1 ДО N
══════════════════════════════════════
n = 5
s = 0
for i in range(1, n + 1):
	s += i
print(s)

# или
print(sum(range(1, 6)))   # 1+2+3+4+5 = 15 — проверь границы в задании!

══════════════════════════════════════
СТРОКИ: len, пробелы, подсчёт без len (идея)
══════════════════════════════════════
s = "hello world"
print(len(s.replace(" ", "")))   # длина без пробелов

# Ручной подсчёт символов без пробелов:
c = 0
for ch in s:
	if ch != " ":
		c += 1
print(c)

══════════════════════════════════════
ТАБЛИЦА УМНОЖЕНИЯ (1..9)
══════════════════════════════════════
# Каждая строка вида: a * b = результат
for a in range(1, 10):
	for b in range(1, 10):
		print(f"{a} * {b} = {a * b}")

ФОРМАТ ВЫВОДА
══════════════════════════════════════

print("sum =", 27)
