PYTHON — Записка 2. Циклы, range, условия, повтор строк
(под задания уровня 1: цепочки чисел, несколько строк, простая логика)

══════════════════════════════════════
RANGE — ДИАПАЗОН ЧИСЕЛ
══════════════════════════════════════
# От 0 до 9 включительно в цикле:
for i in range(10):
	print(i)

# Числа в ОДНУ строку через запятую: 0,1,2,...,9
print(",".join(str(i) for i in range(10)))

# От 5 до 0 с шагом -1 (убывание):
for n in range(5, -1, -1):
	print(n)

══════════════════════════════════════
ОДНА И ТА ЖЕ СТРОКА НЕСКОЛЬКО РАЗ
══════════════════════════════════════
# Три строки «Привет» подряд (каждая с новой строки):
for _ in range(3):
	print("Привет")

# или
print("Привет\n" * 3, end="")   # осторожно: лишний перенос в конце может сломать совпадение

══════════════════════════════════════
УСЛОВИЯ if / elif / else
══════════════════════════════════════
x = 7
if x > 10:
	print("больше 10")
elif x > 5:
	print("больше 5")
else:
	print("5 или меньше")

# Чётность:
if n % 2 == 0:
	print("четное")
else:
	print("нечетное")

# Текущие дата и время на компьютере (формат как в задании):
print(datetime.now())

# Только дата:
print(datetime.now().date())

# Если в задании просят «текущую дату» — используй datetime.now(), не год 2023 из примеров.
